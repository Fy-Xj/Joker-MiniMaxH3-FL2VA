# ---------------------------- 由本工程从源码构建 ----------------------------
# torchvision 0.24.0  (pytorch/vision, tag v0.24.0)
#   环境变量：BUILD_VERSION=0.24.0 MAX_JOBS=32 FORCE_CUDA=1
#             TORCHVISION_USE_NVJPEG=0 TORCHVISION_USE_VIDEO_CODEC=0 TORCHVISION_USE_FFMPEG=0
#   命令：    python setup.py bdist_wheel
#   原因：通用 torchvision wheel 未注册 torchvision::nms，
